   390→        '<span class="text-slate-300">' + esc(label) + '</span> ' +
   391→        '<span class="text-slate-600">— ' + esc(desc) + '</span>';
   392→      logEl.insertBefore(div, logEl.firstChild);

   401→    async function fireAttack(attack) {
   402→      total++;
   403→      updateKpis();
   404→
   405→      const opts = { method: attack.method || 'GET', redirect: 'manual' };
   406→      if (attack.body) {
   407→        opts.headers = { 'Content-Type': 'application/x-www-form-urlencoded' };
   408→        opts.body = attack.body;
   409→      }
   410→      if (attack.ua) {
   411→        opts.headers = opts.headers || {};
   412→        opts.headers['User-Agent'] = attack.ua;
   413→      }
   414→
   415→      try {
   416→        const resp = await fetch(attack.url, opts);
   417→        const status = resp.status;
   418→        const isBlock = (status === 403);
   419→
   420→        if (isBlock) blocked++; else passed++;
   421→        updateKpis();
   422→        logResult(attack.label, attack.desc, status, isBlock);
   423→      } catch (e) {
   424→        // Network error or CORS issue — most likely a redirect that fetch can't follow manually
   425→        passed++;
   426→        updateKpis();
   427→        logResult(attack.label, attack.desc, 'ERR', false);
   428→      }
   429→    }

   507→    // Fire one brute-force attempt with bad credentials.
   508→    // Returns the HTTP status (302 = redirected, 403 = blocked by WAF if payload triggers).
   509→    async function bruteOnce() {
   510→      bruteAttempt++;
   511→      const fakeUser = 'attacker' + bruteAttempt;
   512→      const fakePass = 'wrongpass' + bruteAttempt;
   513→      try {
   514→        const resp = await fetch('/login', {
   515→          method: 'POST',
   516→          headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
   517→          body: 'studentId=' + encodeURIComponent(fakeUser) + '&password=' + encodeURIComponent(fakePass),
   518→          redirect: 'manual', // so we can inspect the redirect target
   519→        });
   520→        // With redirect: 'manual', fetch returns opaqueredirect (status 0) — but
   521→        // we can detect trap vs reject by following manually. Simpler: re-fetch
   522→        // with redirect: 'follow' and check the final URL.
   523→        return resp.status;
   524→      } catch (e) {
   525→        return 'ERR';
   526→      }
   527→    }
   528→
   529→    // Easier path: use XMLHttpRequest-style to actually follow redirects
   530→    // and detect whether we ended up on /dashboard (the trap now serves
   531→    // honeypot content at /dashboard so the attacker doesn't see /trap).
   532→    function bruteFollow() {
   533→      return new Promise((resolve) => {
   534→        const xhr = new XMLHttpRequest();
   535→        xhr.open('POST', '/login');
   536→        xhr.setRequestHeader('Content-Type', 'application/x-www-form-urlencoded');
   537→        xhr.onreadystatechange = function () {
   538→          if (xhr.readyState === 4) {
   539→            // After following redirects, xhr.responseURL is the final URL.
   540→            // Since we used WRONG credentials, landing on /dashboard means
   541→            // we were trapped (wrong creds should never succeed normally).
   542→            const finalUrl = xhr.responseURL || '';
   543→            const endedOnTrap = finalUrl.includes('/dashboard');
   544→            resolve({ endedOnTrap, finalUrl });
   545→          }
   546→        };
   547→        xhr.send('studentId=attacker' + bruteAttempt + '&password=wrongpass' + bruteAttempt);
   548→      });
   549→    }

   551→    async function doBruteStep(stepNum) {
   552→      bruteAttempt = stepNum;
   553→      const result = await bruteFollow();
   554→
   555→      if (result.endedOnTrap) {
   556→        logResult('Login #' + stepNum + ' (wrong creds)', 'Honeypot TRAP triggered (served fake /dashboard)', '302→/dashboard', false);
   557→        bruteStatus(
   558→          '[TRAPPED] Attempt #' + stepNum + ' was trapped! Server lied "Login Successful" — URL bar shows /dashboard (looks totally normal to the attacker).',
   559→          'text-fuchsia-300 font-bold'
   560→        );
   561→        // Open the fake dashboard in a new tab so you can see what the
   562→        // attacker sees. The URL bar will show /dashboard — no hint of /trap.
   563→        const token = result.finalUrl.split('token=')[1] || '';
   564→        window.open('/dashboard' + (token ? '?token=' + token : ''), '_blank');
   565→      } else if (stepNum >= 3) {
   566→        logResult('Login #' + stepNum + ' (wrong creds)', 'Honeypot ARMED — next attempt will trap', '302→/', false);
   567→        bruteStatus(
   568→          '[ARMED] Attempt #' + stepNum + ' crossed the threshold. Honeypot is active for your IP. Click STEP 4 to be trapped.',
   569→          'text-fuchsia-400'
   570→        );
   571→      } else {
   572→        logResult('Login #' + stepNum + ' (wrong creds)', 'Brute-force attempt #' + stepNum + ' — failed', '302→/', false);
   573→        bruteStatus(
   574→          '[PROBE] Attempt #' + stepNum + ' failed normally. Threshold is 3. ' + (3 - stepNum) + ' more to arm the honeypot.',
   575→          'text-amber-400'
   576→        );
   577→      }
   578→    }
