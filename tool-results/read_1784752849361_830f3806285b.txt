   178→    function getToken() {
   179→      const urlToken = new URLSearchParams(window.location.search).get('token');
   180→      if (urlToken) { try { localStorage.setItem('dwaf_token', urlToken); } catch (e) {} return urlToken; }
   181→      try { return localStorage.getItem('dwaf_token'); } catch (e) { return null; }
   182→    }
   183→    function clearToken() { try { localStorage.removeItem('dwaf_token'); } catch (e) {} }
   184→    function esc(s) {
   185→      return String(s == null ? '' : s)
   186→        .replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;')
   187→        .replace(/"/g, '&quot;').replace(/'/g, '&#39;');
   188→    }

   193→    (function wireLogout() {
   194→      const form = document.getElementById('logoutForm');
   195→      if (!form) return;
   196→      form.addEventListener('submit', (e) => {
   197→        e.preventDefault();
   198→        const token = getToken();
   199→        const url = token ? '/logout?token=' + encodeURIComponent(token) : '/logout';
   200→        fetch(url, { method: 'POST', credentials: 'same-origin' })
   201→          .then(() => { clearToken(); window.location.href = '/'; })
   202→          .catch(() => { clearToken(); window.location.href = '/'; });
   203→      });
   204→    })();

   209→    (async function checkSession() {
   210→      const token = getToken();
   211→      if (!token) { window.location.href = '/?error=auth'; return; }
   212→      try {
   213→        const resp = await fetch('/api/me?token=' + encodeURIComponent(token), { credentials: 'same-origin' });
   214→        if (!resp.ok) { clearToken(); window.location.href = '/?error=auth'; return; }
   215→        const data = await resp.json();
   216→        if (data.user.role !== 'admin') window.location.href = '/dashboard?token=' + encodeURIComponent(token);
   217→      } catch (e) { console.error('[soc] session check failed:', e); }
   218→    })();

   232→    function fmtTime(iso) {
   233→      const d = new Date(iso);
   234→      const p = (n) => String(n).padStart(2, '0');
   235→      return p(d.getHours()) + ':' + p(d.getMinutes()) + ':' + p(d.getSeconds());
   236→    }

   238→    // Convert ISO country code to flag emoji
   239→    function flagEmoji(cc) {
   240→      if (!cc || cc.length !== 2 || cc === '??' || cc === 'LO') return '';
   241→      const cp = [...cc.toUpperCase()].map(c => 0x1F1E6 + c.charCodeAt(0) - 65);
   242→      return String.fromCodePoint(...cp) + ' ';
   243→    }
   244→
   245→    function fmtGeo(g) {
   246→      if (!g) return '';
   247→      const flag = flagEmoji(g.countryCode);
   248→      if (g.source === 'local') return flag + 'Local Network';
   249→      if (g.source === 'fallback' || g.source === 'pending') return flag + 'geo...';
   250→      const parts = [];
   251→      if (g.city && g.city !== '—') parts.push(g.city);
   252→      if (g.country && g.country !== 'Unknown') parts.push(g.country);
   253→      return flag + (parts.length ? parts.join(', ') : 'Unknown');
   254→    }
   255→
   256→    function fmtGeoDetail(g) {
   257→      if (!g) return '';
   258→      const lines = [];
   259→      if (g.isp && g.isp !== '—') lines.push('ISP: ' + g.isp);
   260→      if (g.org && g.org !== '—') lines.push('Org: ' + g.org);
   261→      if (g.source) lines.push('source: ' + g.source);
   262→      return lines.join(' / ');
   263→    }

   396→    function updateAttackerCatalog(evt) {
   397→      if (!evt.ip) return;
   398→      let a = attackers.get(evt.ip);
   399→      if (!a) {
   400→        a = { ip: evt.ip, geo: evt.geo || null, total: 0, blocked: 0, flagged: 0, trapped: 0, failed: 0, lastSeen: evt.timestamp };
   401→        attackers.set(evt.ip, a);
   402→      }
   403→      a.total++;
   404→      a.lastSeen = evt.timestamp;
   405→      if (evt.geo) a.geo = evt.geo;
   406→      if (evt.kind === 'waf' && evt.type === 'block') a.blocked++;
   407→      else if (evt.kind === 'waf' && evt.type === 'flag') a.flagged++;
   408→      else if (evt.kind === 'honeypot' && evt.type === 'trapped') a.trapped++;
   409→      else if (evt.kind === 'honeypot' && evt.type === 'failed') a.failed++;
   410→      renderTopAttackers();
   411→    }

   473→    async function loadHistory() {
   474→      const token = getToken();
   475→      if (!token) return;
   476→      try {
   477→        const resp = await fetch('/api/events/recent?limit=50&token=' + encodeURIComponent(token), { credentials: 'same-origin' });
   478→        if (!resp.ok) {
   479→          document.getElementById('histStatus').textContent = 'failed (' + resp.status + ')';
   480→          document.getElementById('histStatus').className = 'text-red-400';
   481→          return;
   482→        }
   483→        const data = await resp.json();
   484→        const events = (data.events || []).slice().reverse(); // oldest first for append
   485→        if (events.length === 0) {
   486→          feed.innerHTML = '<div class="text-slate-500">[boot] No events in history yet.</div>' +
   487→            '<div class="text-slate-600 mt-2"><span class="text-green-700">&gt;&gt;</span> Open <a href="/test" class="text-green-400 underline">/test</a> and fire attacks to populate this feed.</div>' +
   488→            '<div class="text-slate-700 mt-4">[idle] listening...<span class="blink">_</span></div>';
   489→        } else {
   490→          feed.innerHTML = '<div class="text-slate-500">[history] loaded ' + events.length + ' recent events:</div>';
   491→          for (const evt of events) {
   492→            if (evt.kind === 'waf') renderWafEvent(evt);
   493→            else if (evt.kind === 'honeypot') renderHoneypotEvent(evt);
   494→          }
   495→          appendFeed('<div class="text-green-500 mt-2">--- end of history, listening live ---</div>');
   496→        }
   497→        document.getElementById('histStatus').textContent = 'loaded (' + events.length + ')';
   498→        document.getElementById('histStatus').className = 'text-green-400';
   499→
   500→        // Also load top attackers
   501→        const aResp = await fetch('/api/attackers/top?limit=20&token=' + encodeURIComponent(token), { credentials: 'same-origin' });
   502→        if (aResp.ok) {
   503→          const aData = await aResp.json();
   504→          for (const a of (aData.attackers || [])) {
   505→            attackers.set(a.ip, {
   506→              ip: a.ip, geo: a.geo, total: a.totalEvents,
   507→              blocked: a.blockedCount, flagged: a.flaggedCount,
   508→              trapped: a.trappedCount, failed: a.failedCount,
   509→              lastSeen: a.lastSeen,
   510→            });
   511→          }
   512→          renderTopAttackers();
   513→        }
   514→
   515→        // Load aggregate stats
   516→        const sResp = await fetch('/api/stats?token=' + encodeURIComponent(token), { credentials: 'same-origin' });
   517→        if (sResp.ok) {
   518→          const sData = await sResp.json();
   519→          blockedCount = sData.blocked || 0;
   520→          trappedCount = sData.trapped || 0;
   521→          geoCacheSize = sData.geo?.cacheSize || 0;
   522→          // uniqueIps from attackers map
   523→          for (const ip of attackers.keys()) uniqueIps.add(ip);
   524→          updateKpis();
   525→        }
   526→      } catch (e) {
   527→        console.error('[soc] loadHistory error:', e);
   528→        document.getElementById('histStatus').textContent = 'error';
   529→        document.getElementById('histStatus').className = 'text-red-400';
   530→      }
   531→    }
