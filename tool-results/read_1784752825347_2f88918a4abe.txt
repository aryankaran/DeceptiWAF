   108→// Try to URL-decode a string. Failures return the input unchanged.
   109→function safeDecode(s) {
   110→  if (typeof s !== 'string') return s;
   111→  try { return decodeURIComponent(s); } catch { return s; }
   112→}
   113→
   114→// Recursively URL-decode until stable (catches double-encoded payloads
   115→// like %2527 — encoded twice to evade single-decode WAFs).
   116→function deepDecode(s, maxPasses = 3) {
   117→  if (typeof s !== 'string') return s;
   118→  let prev = s;
   119→  for (let i = 0; i < maxPasses; i++) {
   120→    const next = safeDecode(prev);
   121→    if (next === prev) break;
   122→    prev = next;
   123→  }
   124→  return prev;
   125→}

   127→// HTML entity decoder — handles named, decimal, and hex entities.
   128→// Catches &lt;script&gt;, &#60;script&#62;, &#x3c;script&#x3e; etc.
   129→const NAMED_ENTITIES = {
   130→  lt: '<', gt: '>', amp: '&', quot: '"', apos: "'", nbsp: ' ',
   131→  tab: '\t', newline: '\n', copy: '©', reg: '®', trade: '™',
   132→  ldquo: '"', rdquo: '"', lsquo: "'", rsquo: "'",
   133→  sol: '/', verbar: '|', num: '#', cent: '¢', pound: '£', yen: '¥',
   134→  euro: '€',sect: '§', deg: '°', plusmn: '±', times: '×', divide: '÷',
   135→};
   136→
   137→function decodeHtmlEntities(s) {
   138→  if (typeof s !== 'string') return s;
   139→  return s.replace(/&(#[xX]?[0-9a-fA-F]+|[a-zA-Z][a-zA-Z0-9]*);/g, (m, ent) => {
   140→    if (ent[0] === '#') {
   141→      let cp;
   142→      if (ent[1] === 'x' || ent[1] === 'X') {
   143→        cp = parseInt(ent.slice(2), 16);
   144→      } else {
   145→        cp = parseInt(ent.slice(1), 10);
   146→      }
   147→      if (isNaN(cp) || cp < 0 || cp > 0x10FFFF) return m;
   148→      try { return String.fromCodePoint(cp); } catch { return m; }
   149→    } else {
   150→      return NAMED_ENTITIES[ent.toLowerCase()] || m;
   151→    }
   152→  });
   153→}

   155→// Unicode NFKC normalization — converts full-width chars (ＵＮＩＯＮ) and
   156→// other compatibility forms to their ASCII equivalents. Catches
   157→// 'ＵＮＩＯＮ ＳＥＬＥＣＴ' style evasion.
   158→function normalizeUnicode(s) {
   159→  if (typeof s !== 'string') return s;
   160→  try { return s.normalize('NFKC'); } catch { return s; }
   161→}
   162→
   163→// SQL/whitespace normalizer — replaces inline comments with spaces and
   164→// collapses runs of whitespace. Catches 'UNION/**/SELECT' style evasion
   165→// and 'UNION   SELECT' padding.
   166→function normalizeForSql(s) {
   167→  if (typeof s !== 'string') return s;
   168→  return s
   169→    .replace(/\/\*[\s\S]*?\*\//g, ' ')  // /* comment */ -> space
   170→    .replace(/\s+/g, ' ')                  // collapse whitespace
   171→    .trim();
   172→}

   174→// Build the full haystack set: every reasonable de-obfuscation layer
   175→// applied in sequence, deduplicated. The WAF tests each pattern against
   176→// EACH haystack so layered obfuscation can't slip through.
   177→function buildHaystacks(raw, opts = {}) {
   178→  if (typeof raw !== 'string' || raw.length === 0) return [];
   179→  const decode = opts.decode !== false;
   180→  const haystacks = new Set();
   181→
   182→  haystacks.add(raw);
   183→
   184→  if (decode) {
   185→    const urlDecoded = deepDecode(raw);
   186→    haystacks.add(urlDecoded);
   187→  }
   188→
   189→  // Apply HTML entity decoding to raw and urlDecoded
   190→  const htmlRaw = decodeHtmlEntities(raw);
   191→  haystacks.add(htmlRaw);
   192→  if (decode) {
   193→    const htmlUrl = decodeHtmlEntities(deepDecode(raw));
   194→    haystacks.add(htmlUrl);
   195→  }
   196→
   197→  // Apply Unicode normalization to everything we have so far
   198→  const uniSet = new Set();
   199→  for (const h of haystacks) {
   200→    const n = normalizeUnicode(h);
   201→    if (n !== h) uniSet.add(n);
   202→  }
   203→  for (const u of uniSet) haystacks.add(u);
   204→
   205→  // Apply SQL normalization to everything (catches UNION/**/SELECT)
   206→  const sqlSet = new Set();
   207→  for (const h of haystacks) {
   208→    const n = normalizeForSql(h);
   209→    if (n !== h) sqlSet.add(n);
   210→  }
   211→  for (const s of sqlSet) haystacks.add(s);
   212→
   213→  return Array.from(haystacks);
   214→}

   216→// ------------------------------------------------------------------
   217→// Inspection — collect every interesting string from the request
   218→// ------------------------------------------------------------------
   219→function collectInputVectors(req) {
   220→  const vectors = [];
   221→
   222→  // 1. URL path
   223→  if (req.path) vectors.push({ source: 'path', value: req.path });
   224→
   225→  // 2. Query string values
   226→  if (req.query && typeof req.query === 'object') {
   227→    for (const [k, v] of Object.entries(req.query)) {
   228→      if (k === 'token') continue; // never inspect session tokens
   229→      if (typeof v === 'string') vectors.push({ source: `query.${k}`, value: v });
   230→    }
   231→  }
   232→
   233→  // 3. Body fields (urlencoded or JSON)
   234→  if (req.body && typeof req.body === 'object') {
   235→    for (const [k, v] of Object.entries(req.body)) {
   236→      if (typeof v === 'string') vectors.push({ source: `body.${k}`, value: v });
   237→    }
   238→  }
   239→
   240→  // 4. Headers worth inspecting
   241→  const ua = req.get('user-agent');
   242→  if (ua) vectors.push({ source: 'header.user-agent', value: ua, decode: false });
   243→
   244→  const ref = req.get('referer');
   245→  if (ref) vectors.push({ source: 'header.referer', value: ref });
   246→
   247→  return vectors;
   248→}

   250→// ------------------------------------------------------------------
   251→// Core: inspect a single string against the pattern dictionary
   252→// ------------------------------------------------------------------
   253→function scanString(raw, opts = {}) {
   254→  // Phase 5: build a multi-layer haystack so layered obfuscation
   255→  // (URL-encode + HTML-entity + Unicode full-width + SQL comments)
   256→  // can't slip through. Each pattern is tested against each haystack.
   257→  const haystacks = buildHaystacks(raw, opts);
   258→
   259→  const hits = [];
   260→  for (const pattern of PATTERNS) {
   261→    for (const h of haystacks) {
   262→      if (pattern.pattern.test(h)) {
   263→        hits.push({
   264→          id: pattern.id,
   265→          family: pattern.family,
   266→          label: pattern.label,
   267→          score: pattern.score,
   268→          source: opts.source || 'unknown',
   269→        });
   270→        break; // don't double-count the same pattern across haystacks
   271→      }
   272→    }
   273→  }
   274→  return hits;
   275→}

   277→// ------------------------------------------------------------------
   278→// Inspect the whole request — returns the analysis object
   279→// ------------------------------------------------------------------
   280→function inspectRequest(req) {
   281→  const vectors = collectInputVectors(req);
   282→  const allHits = [];
   283→  let totalScore = 0;
   284→
   285→  for (const vec of vectors) {
   286→    const hits = scanString(vec.value, { source: vec.source, decode: vec.decode !== false });
   287→    for (const h of hits) {
   288→      allHits.push({ ...h, source: vec.source, sample: vec.value.slice(0, 200) });
   289→      totalScore += h.score;
   290→    }
   291→  }
   292→
   293→  // Deduplicate hits by pattern id+source (so multiple haystacks don't inflate)
   294→  const seen = new Set();
   295→  const deduped = [];
   296→  for (const h of allHits) {
   297→    const key = `${h.id}@${h.source}`;
   298→    if (seen.has(key)) continue;
   299→    seen.add(key);
   300→    deduped.push(h);
   301→  }
   302→  // Recompute score on deduped
   303→  totalScore = deduped.reduce((s, h) => s + h.score, 0);
   304→
   305→  // Determine primary attack family (highest-scoring family)
   306→  const familyScores = {};
   307→  for (const h of deduped) {
   308→    familyScores[h.family] = (familyScores[h.family] || 0) + h.score;
   309→  }
   310→  const topFamily = Object.entries(familyScores).sort((a, b) => b[1] - a[1])[0];
   311→  const attackType = topFamily ? topFamily[0] : 'Unknown';
   312→
   313→  return {
   314→    blocked: totalScore > BLOCK_THRESHOLD,
   315→    score: totalScore,
   316→    threshold: BLOCK_THRESHOLD,
   317→    attackType,
   318→    hits: deduped,
   319→    ip: req.ip,
   320→    method: req.method,
   321→    path: req.path,
   322→    userAgent: req.get('user-agent') || '',
   323→    // Phase 5+ fingerprinting — capture browser/network metadata for SOC display
   324→    fingerprint: {
   325→      userAgent: req.get('user-agent') || '',
   326→      acceptLanguage: req.get('accept-language') || '',
   327→      acceptEncoding: req.get('accept-encoding') || '',
   328→      secChUa: req.get('sec-ch-ua') || '',
   329→      secChUaPlatform: req.get('sec-ch-ua-platform') || '',
   330→      secChUaMobile: req.get('sec-ch-ua-mobile') || '',
   331→      secFetchMode: req.get('sec-fetch-mode') || '',
   332→      secFetchSite: req.get('sec-fetch-site') || '',
   333→      referer: req.get('referer') || '',
   334→      host: req.get('host') || '',
   335→      connection: req.get('connection') || '',
   336→    },
   337→    timestamp: new Date().toISOString(),
   338→  };
   339→}

   341→// ------------------------------------------------------------------
   342→// Express middleware
   343→// ------------------------------------------------------------------
   344→function wafMiddleware(req, res, next) {
   345→  // Skip inspection for static asset requests (favicon, etc.) — keep
   346→  // logs clean and reduce noise.
   347→  if (req.path.match(/\.(ico|png|jpg|jpeg|gif|css|js|map)$/i)) {
   348→    return next();
   349→  }
   350→
   351→  const result = inspectRequest(req);
   352→
   353→  if (result.blocked) {
   354→    // ---- Block path ----
   355→    const io = req.app.get('io');
   356→    const eventPayload = {
   357→      kind: 'waf',
   358→      type: 'block',
   359→      attackType: result.attackType,
   360→      score: result.score,
   361→      threshold: result.threshold,
   362→      ip: result.ip,
   363→      method: result.method,
   364→      path: result.path,
   365→      hits: result.hits,
   366→      userAgent: result.userAgent,
   367→      timestamp: result.timestamp,
   368→    };
   369→
   370→    if (io) io.emit('waf_event', eventPayload);
   371→
   372→    // Phase 4: store in event history (best-effort geo lookup, async)
   373→    storeEventWithGeo(eventPayload);
   374→
   375→    // ANSI-coloured console log
   376→    const red = '\x1b[31m', yellow = '\x1b[33m', cyan = '\x1b[36m', dim = '\x1b[2m', reset = '\x1b[0m';
   377→    console.log(
   378→      `${red}[WAF BLOCK]${reset} ${cyan}${result.attackType}${reset} ` +
   379→      `score=${red}${result.score}${reset}/${result.threshold} ` +
   380→      `ip=${result.ip} ${dim}${result.method} ${result.path}${reset}`
   381→    );
   382→    for (const h of result.hits) {
   383→      console.log(`  ${yellow}-${reset} ${h.family}/${h.id} (${h.score}pt) source=${h.source} ${dim}${h.label}${reset}`);
   384→    }
   385→
   386→    // Inline 403 page (no template engine — keeps the project build-free)
   387→    return res.status(403).type('html').send(renderBlockedPage(result));
   388→  }
   389→
   390→  if (result.score > 0) {
   391→    // ---- Suspicious-but-allowed path ----
   392→    const io = req.app.get('io');
   393→    const eventPayload = {
   394→      kind: 'waf',
   395→      type: 'flag',
   396→      attackType: result.attackType,
   397→      score: result.score,
   398→      threshold: result.threshold,
   399→      ip: result.ip,
   400→      method: result.method,
   401→      path: result.path,
   402→      hits: result.hits,
   403→      userAgent: result.userAgent,
   404→      timestamp: result.timestamp,
   405→    };
   406→    if (io) io.emit('waf_event', eventPayload);
   407→    storeEventWithGeo(eventPayload);
   408→
   409→    const yellow = '\x1b[33m', dim = '\x1b[2m', reset = '\x1b[0m';
