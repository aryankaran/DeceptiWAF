   108→   108→// Try to URL-decode a string. Failures return the input unchanged.
   109→   109→function safeDecode(s) {
   110→   110→  if (typeof s !== 'string') return s;
   111→   111→  try { return decodeURIComponent(s); } catch { return s; }
   112→   112→}
   113→   113→
   114→   114→// Recursively URL-decode until stable (catches double-encoded payloads
   115→   115→// like %2527 — encoded twice to evade single-decode WAFs).
   116→   116→function deepDecode(s, maxPasses = 3) {
   117→   117→  if (typeof s !== 'string') return s;
   118→   118→  let prev = s;
   119→   119→  for (let i = 0; i < maxPasses; i++) {
   120→   120→    const next = safeDecode(prev);
   121→   121→    if (next === prev) break;
   122→   122→    prev = next;
   123→   123→  }
   124→   124→  return prev;
   125→   125→}

   127→   127→// HTML entity decoder — handles named, decimal, and hex entities.
   128→   128→// Catches &lt;script&gt;, &#60;script&#62;, &#x3c;script&#x3e; etc.
   129→   129→const NAMED_ENTITIES = {
   130→   130→  lt: '<', gt: '>', amp: '&', quot: '"', apos: "'", nbsp: ' ',
   131→   131→  tab: '\t', newline: '\n', copy: '©', reg: '®', trade: '™',
   132→   132→  ldquo: '"', rdquo: '"', lsquo: "'", rsquo: "'",
   133→   133→  sol: '/', verbar: '|', num: '#', cent: '¢', pound: '£', yen: '¥',
   134→   134→  euro: '€',sect: '§', deg: '°', plusmn: '±', times: '×', divide: '÷',
   135→   135→};
   136→   136→
   137→   137→function decodeHtmlEntities(s) {
   138→   138→  if (typeof s !== 'string') return s;
   139→   139→  return s.replace(/&(#[xX]?[0-9a-fA-F]+|[a-zA-Z][a-zA-Z0-9]*);/g, (m, ent) => {
   140→   140→    if (ent[0] === '#') {
   141→   141→      let cp;
   142→   142→      if (ent[1] === 'x' || ent[1] === 'X') {
   143→   143→        cp = parseInt(ent.slice(2), 16);
   144→   144→      } else {
   145→   145→        cp = parseInt(ent.slice(1), 10);
   146→   146→      }
   147→   147→      if (isNaN(cp) || cp < 0 || cp > 0x10FFFF) return m;
   148→   148→      try { return String.fromCodePoint(cp); } catch { return m; }
   149→   149→    } else {
   150→   150→      return NAMED_ENTITIES[ent.toLowerCase()] || m;
   151→   151→    }
   152→   152→  });
   153→   153→}

   155→   155→// Unicode NFKC normalization — converts full-width chars (ＵＮＩＯＮ) and
   156→   156→// other compatibility forms to their ASCII equivalents. Catches
   157→   157→// 'ＵＮＩＯＮ ＳＥＬＥＣＴ' style evasion.
   158→   158→function normalizeUnicode(s) {
   159→   159→  if (typeof s !== 'string') return s;
   160→   160→  try { return s.normalize('NFKC'); } catch { return s; }
   161→   161→}
   162→   162→
   163→   163→// SQL/whitespace normalizer — replaces inline comments with spaces and
   164→   164→// collapses runs of whitespace. Catches 'UNION/**/SELECT' style evasion
   165→   165→// and 'UNION   SELECT' padding.
   166→   166→function normalizeForSql(s) {
   167→   167→  if (typeof s !== 'string') return s;
   168→   168→  return s
   169→   169→    .replace(/\/\*[\s\S]*?\*\//g, ' ')  // /* comment */ -> space
   170→   170→    .replace(/\s+/g, ' ')                  // collapse whitespace
   171→   171→    .trim();
   172→   172→}
   173→   173→
   174→   174→// Build the full haystack set: every reasonable de-obfuscation layer
   175→   175→// applied in sequence, deduplicated. The WAF tests each pattern against
   176→   176→// EACH haystack so layered obfuscation can't slip through.
   177→   177→function buildHaystacks(raw, opts = {}) {
   178→   178→  if (typeof raw !== 'string' || raw.length === 0) return [];
   179→   179→  const decode = opts.decode !== false;
   180→   180→  const haystacks = new Set();
   181→   181→
   182→   182→  haystacks.add(raw);
   183→   183→
   184→   184→  if (decode) {
   185→   185→    const urlDecoded = deepDecode(raw);
   186→   186→    haystacks.add(urlDecoded);
   187→   187→  }
   188→   188→
   189→   189→  // Apply HTML entity decoding to raw and urlDecoded
   190→   190→  const htmlRaw = decodeHtmlEntities(raw);
   191→   191→  haystacks.add(htmlRaw);
   192→   192→  if (decode) {
   193→   193→    const htmlUrl = decodeHtmlEntities(deepDecode(raw));
   194→   194→    haystacks.add(htmlUrl);
   195→   195→  }
   196→   196→
   197→   197→  // Apply Unicode normalization to everything we have so far
   198→   198→  const uniSet = new Set();
   199→   199→  for (const h of haystacks) {
   200→   200→    const n = normalizeUnicode(h);
   201→   201→    if (n !== h) uniSet.add(n);
   202→   202→  }
   203→   203→  for (const u of uniSet) haystacks.add(u);
   204→   204→
   205→   205→  // Apply SQL normalization to everything (catches UNION/**/SELECT)
   206→   206→  const sqlSet = new Set();
   207→   207→  for (const h of haystacks) {
   208→   208→    const n = normalizeForSql(h);
   209→   209→    if (n !== h) sqlSet.add(n);
   210→   210→  }
   211→   211→  for (const s of sqlSet) haystacks.add(s);
   212→   212→
   213→   213→  return Array.from(haystacks);
   214→   214→}

   216→   216→// ------------------------------------------------------------------
   217→   217→// Inspection — collect every interesting string from the request
   218→   218→// ------------------------------------------------------------------
   219→   219→function collectInputVectors(req) {
   220→   220→  const vectors = [];
   221→   221→
   222→   222→  // 1. URL path
   223→   223→  if (req.path) vectors.push({ source: 'path', value: req.path });
   224→   224→
   225→   225→  // 2. Query string values
   226→   226→  if (req.query && typeof req.query === 'object') {
   227→   227→    for (const [k, v] of Object.entries(req.query)) {
   228→   228→      if (k === 'token') continue; // never inspect session tokens
   229→   229→      if (typeof v === 'string') vectors.push({ source: `query.${k}`, value: v });
   230→   230→    }
   231→   231→  }
   232→   232→
   233→   233→  // 3. Body fields (urlencoded or JSON)
   234→   234→  if (req.body && typeof req.body === 'object') {
   235→   235→    for (const [k, v] of Object.entries(req.body)) {
   236→   236→      if (typeof v === 'string') vectors.push({ source: `body.${k}`, value: v });
   237→   237→    }
   238→   238→  }
   239→   239→
   240→   240→  // 4. Headers worth inspecting
   241→   241→  const ua = req.get('user-agent');
   242→   242→  if (ua) vectors.push({ source: 'header.user-agent', value: ua, decode: false });
   243→   243→
   244→   244→  const ref = req.get('referer');
   245→   245→  if (ref) vectors.push({ source: 'header.referer', value: ref });
   246→   246→
   247→   247→  return vectors;
   248→   248→}

   250→   250→// ------------------------------------------------------------------
   251→   251→// Core: inspect a single string against the pattern dictionary
   252→   252→// ------------------------------------------------------------------
   253→   253→function scanString(raw, opts = {}) {
   254→   254→  // Phase 5: build a multi-layer haystack so layered obfuscation
   255→   255→  // (URL-encode + HTML-entity + Unicode full-width + SQL comments)
   256→   256→  // can't slip through. Each pattern is tested against each haystack.
   257→   257→  const haystacks = buildHaystacks(raw, opts);
   258→   258→
   259→   259→  const hits = [];
   260→   260→  for (const pattern of PATTERNS) {
   261→   261→    for (const h of haystacks) {
   262→   262→      if (pattern.pattern.test(h)) {
   263→   263→        hits.push({
   264→   264→          id: pattern.id,
   265→   265→          family: pattern.family,
   266→   266→          label: pattern.label,
   267→   267→          score: pattern.score,
   268→   268→          source: opts.source || 'unknown',
   269→   269→        });
   270→   270→        break; // don't double-count the same pattern across haystacks
   271→   271→      }
   272→   272→    }
   273→   273→  }
   274→   274→  return hits;
   275→   275→}

   277→   277→// ------------------------------------------------------------------
   278→   278→// Inspect the whole request — returns the analysis object
   279→   279→// ------------------------------------------------------------------
   280→   280→function inspectRequest(req) {
   281→   281→  const vectors = collectInputVectors(req);
   282→   282→  const allHits = [];
   283→   283→  let totalScore = 0;
   284→   284→
   285→   285→  for (const vec of vectors) {
   286→   286→    const hits = scanString(vec.value, { source: vec.source, decode: vec.decode !== false });
   287→   287→    for (const h of hits) {
   288→   288→      allHits.push({ ...h, source: vec.source, sample: vec.value.slice(0, 200) });
   289→   289→      totalScore += h.score;
   290→   290→    }
   291→   291→  }
   292→   292→
   293→   293→  // Deduplicate hits by pattern id+source (so multiple haystacks don't inflate)
   294→   294→  const seen = new Set();
   295→   295→  const deduped = [];
   296→   296→  for (const h of allHits) {
   297→   297→    const key = `${h.id}@${h.source}`;
   298→   298→    if (seen.has(key)) continue;
   299→   299→    seen.add(key);
   300→   300→    deduped.push(h);
   301→   301→  }
   302→   302→  // Recompute score on deduped
   303→   303→  totalScore = deduped.reduce((s, h) => s + h.score, 0);
   304→   304→
   305→   305→  // Determine primary attack family (highest-scoring family)
   306→   306→  const familyScores = {};
   307→   307→  for (const h of deduped) {
   308→   308→    familyScores[h.family] = (familyScores[h.family] || 0) + h.score;
   309→   309→  }
   310→   310→  const topFamily = Object.entries(familyScores).sort((a, b) => b[1] - a[1])[0];
   311→   311→  const attackType = topFamily ? topFamily[0] : 'Unknown';
   312→   312→
   313→   313→  return {
   314→   314→    blocked: totalScore > BLOCK_THRESHOLD,
   315→   315→    score: totalScore,
   316→   316→    threshold: BLOCK_THRESHOLD,
   317→   317→    attackType,
   318→   318→    hits: deduped,
   319→   319→    ip: req.ip,
   320→   320→    method: req.method,
   321→   321→    path: req.path,
   322→   322→    userAgent: req.get('user-agent') || '',
   323→   323→    // Phase 5+ fingerprinting — capture browser/network metadata for SOC display
   324→   324→    fingerprint: {
   325→   325→      userAgent: req.get('user-agent') || '',
   326→   326→      acceptLanguage: req.get('accept-language') || '',
   327→   327→      acceptEncoding: req.get('accept-encoding') || '',
   328→   328→      secChUa: req.get('sec-ch-ua') || '',
   329→   329→      secChUaPlatform: req.get('sec-ch-ua-platform') || '',
   330→   330→      secChUaMobile: req.get('sec-ch-ua-mobile') || '',
   331→   331→      secFetchMode: req.get('sec-fetch-mode') || '',
   332→   332→      secFetchSite: req.get('sec-fetch-site') || '',
   333→   333→      referer: req.get('referer') || '',
   334→   334→      host: req.get('host') || '',
   335→   335→      connection: req.get('connection') || '',
   336→   336→    },
   337→   337→    timestamp: new Date().toISOString(),
   338→   338→  };
   339→   339→}

   342→   342→// Express middleware
   343→   343→// ------------------------------------------------------------------
   344→   344→function wafMiddleware(req, res, next) {
   345→   345→  // Skip inspection for static asset requests (favicon, etc.) — keep
   346→   346→  // logs clean and reduce noise.
   347→   347→  if (req.path.match(/\.(ico|png|jpg|jpeg|gif|css|js|map)$/i)) {
   348→   348→    return next();
   349→   349→  }
   350→   350→
   351→   351→  const result = inspectRequest(req);
   352→   352→
   353→   353→  if (result.blocked) {
   354→   354→    // ---- Block path ----
   355→   355→    const io = req.app.get('io');
   356→   356→    const eventPayload = {
   357→   357→      kind: 'waf',
   358→   358→      type: 'block',
   359→   359→      attackType: result.attackType,
   360→   360→      score: result.score,
   361→   361→      threshold: result.threshold,
   362→   362→      ip: result.ip,
   363→   363→      method: result.method,
   364→   364→      path: result.path,
   365→   365→      hits: result.hits,
   366→   366→      userAgent: result.userAgent,
   367→   367→      timestamp: result.timestamp,
   368→   368→    };
   369→   369→
   370→   370→    if (io) io.emit('waf_event', eventPayload);
   371→   371→
   372→   372→    // Phase 4: store in event history (best-effort geo lookup, async)
   373→   373→    storeEventWithGeo(eventPayload);
   374→   374→
   375→   375→    // ANSI-coloured console log
   376→   376→    const red = '\x1b[31m', yellow = '\x1b[33m', cyan = '\x1b[36m', dim = '\x1b[2m', reset = '\x1b[0m';
   377→   377→    console.log(
   378→   378→      `${red}[WAF BLOCK]${reset} ${cyan}${result.attackType}${reset} ` +
   379→   379→      `score=${red}${result.score}${reset}/${result.threshold} ` +
   380→   380→      `ip=${result.ip} ${dim}${result.method} ${result.path}${reset}`
   381→   381→    );
   382→   382→    for (const h of result.hits) {
   383→   383→      console.log(`  ${yellow}-${reset} ${h.family}/${h.id} (${h.score}pt) source=${h.source} ${dim}${h.label}${reset}`);
   384→   384→    }
   385→   385→
   386→   386→    // Inline 403 page (no template engine — keeps the project build-free)
   387→   387→    return res.status(403).type('html').send(renderBlockedPage(result));
   388→   388→  }
   389→   389→
   390→   390→  if (result.score > 0) {
   391→   391→    // ---- Suspicious-but-allowed path ----
   392→   392→    const io = req.app.get('io');
   393→   393→    const eventPayload = {
   394→   394→      kind: 'waf',
   395→   395→      type: 'flag',
   396→   396→      attackType: result.attackType,
   397→   397→      score: result.score,
   398→   398→      threshold: result.threshold,
   399→   399→      ip: result.ip,
   400→   400→      method: result.method,
   401→   401→      path: result.path,
   402→   402→      hits: result.hits,
   403→   403→      userAgent: result.userAgent,
   404→   404→      timestamp: result.timestamp,
   405→   405→    };
   406→   406→    if (io) io.emit('waf_event', eventPayload);
   407→   407→    storeEventWithGeo(eventPayload);
   408→   408→
   409→   409→    const yellow = '\x1b[33m', dim = '\x1b[2m', reset = '\x1b[0m';
   410→   410→    console.log(
   411→   411→      `${yellow}[WAF FLAG]${reset} score=${result.score}/${result.threshold} ` +
   412→   412→      `ip=${result.ip} ${dim}${result.method} ${result.path}${reset}`
   413→   413→    );
   414→   414→    for (const h of result.hits) {
   415→   415→      console.log(`  ${yellow}-${reset} ${h.family}/${h.id} (${h.score}pt) source=${h.source} ${dim}${h.label}${reset}`);
   416→   416→    }
   417→   417→  }
   418→   418→
   419→   419→  return next();
   420→   420→}

   422→   422→// ------------------------------------------------------------------
   423→   423→// Phase 4 helper: store event with async geo lookup
   424→   424→// ------------------------------------------------------------------
   425→   425→let _eventStore = null;
   426→   426→let _geo = null;
   427→   427→function lazyDeps() {
   428→   428→  if (!_eventStore) _eventStore = require('./eventStore');
   429→   429→  if (!_geo) _geo = require('./geo');
   430→   430→}
   431→   431→function storeEventWithGeo(payload) {
   432→   432→  try {
   433→   433→    lazyDeps();
   434→   434→    // Synchronous store first (with cached/sync geo if available)
   435→   435→    const syncGeo = _geo.lookupSync(payload.ip);
   436→   436→    _eventStore.push({ ...payload, geo: syncGeo });
   437→   437→    // If geo is pending, do async lookup and patch the latest entry for that IP
   438→   438→    if (syncGeo.source === 'pending') {
   439→   439→      _geo.lookup(payload.ip).then((fullGeo) => {
   440→   440→        // Patch the attacker catalog with full geo
   441→   441→        // (eventStore.push already created the entry; we just update geo)
   442→   442→        const attackers = _eventStore.topAttackers(1000);
   443→   443→        // We don't have a "patch" API, but the next event from this IP
   444→   444→        // will pick up the cached geo. Good enough for the demo.
   445→   445→      }).catch(() => {});
   446→   446→    }
   447→   447→  } catch (e) {
   448→   448→    console.error('[WAF] storeEventWithGeo error:', e.message);
   449→   449→  }
   450→   450→}
